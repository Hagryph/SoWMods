cmake_minimum_required(VERSION 3.21)
project(Uruks VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

list(APPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../SoWLoader/build/vcpkg_installed/x64-windows-static-md")
find_package(minhook CONFIG REQUIRED)
add_library(Uruks SHARED src/Uruks.cpp src/Settings.cpp src/RecruitmentHooks.cpp src/CaptainGeneration.cpp src/GeneratorNative.cpp)
add_custom_target(CheckHeadlessMods COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/HeadlessMods.cmake" VERBATIM)
add_dependencies(Uruks CheckHeadlessMods)
target_include_directories(Uruks PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../shared")
target_compile_definitions(Uruks PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
target_compile_options(Uruks PRIVATE /utf-8)
target_link_libraries(Uruks PRIVATE minhook::minhook)
set_target_properties(Uruks PROPERTIES PREFIX "")

include(CTest)
if(BUILD_TESTING)
    add_executable(UruksRecruitmentTests tests/RecruitmentTests.cpp)
    target_include_directories(UruksRecruitmentTests PRIVATE src)
    add_test(NAME UruksRecruitmentPolicy COMMAND UruksRecruitmentTests)
    add_executable(CaptainGenerationTests tests/CaptainGenerationTests.cpp src/GeneratorNative.cpp)
    target_link_libraries(CaptainGenerationTests PRIVATE minhook::minhook)
    target_include_directories(CaptainGenerationTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../shared")
    target_compile_definitions(CaptainGenerationTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
    add_test(NAME CaptainGenerationSnapshots COMMAND CaptainGenerationTests)
endif()
