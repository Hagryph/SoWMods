cmake_minimum_required(VERSION 3.21)
project(UrukEditor VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

list(APPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../SoWLoader/build/vcpkg_installed/x64-windows-static-md")
find_package(minhook CONFIG REQUIRED)
add_library(UrukEditor SHARED src/UrukEditor.cpp src/Settings.cpp src/RecruitmentHooks.cpp)
target_include_directories(UrukEditor PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../shared")
target_compile_definitions(UrukEditor PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
target_compile_options(UrukEditor PRIVATE /utf-8)
target_link_libraries(UrukEditor PRIVATE minhook::minhook)
set_target_properties(UrukEditor PROPERTIES PREFIX "")

include(CTest)
if(BUILD_TESTING)
    add_executable(UrukRecruitmentTests tests/RecruitmentTests.cpp)
    target_include_directories(UrukRecruitmentTests PRIVATE src)
    add_test(NAME UrukRecruitmentPolicy COMMAND UrukRecruitmentTests)
    add_executable(UrukSettingsTests tests/SettingsTests.cpp src/Settings.cpp)
    target_include_directories(UrukSettingsTests PRIVATE src)
    target_compile_definitions(UrukSettingsTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
    set_target_properties(UrukSettingsTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/settings-test")
    add_test(NAME UrukSettingsPersistence COMMAND UrukSettingsTests)
endif()
